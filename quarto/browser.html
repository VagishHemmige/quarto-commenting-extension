3. Browser script

Author

Vagish Hemmige

The script runs in the reader’s browser. It shows the 💬 Comment button next to a selection, runs the comment dialog, and sends the comment to the relay. It has no dependencies, and the whole file is wrapped in a function so that nothing except what the filter set (window.ISSUE_COMMENTS) is shared with the page.

NoteSource code

_extensions/issue-comments/issue-comments.js

Helpers it shares with the relay: issueTitle must match buildIssue_ in apps-script/Code.gs.

Helpers

store reads or saves one value in localStorage. It is wrapped in try because some browsers throw when storage is blocked (private windows, strict privacy settings); the script then simply doesn’t remember the name.

_extensions/issue-comments/issue-comments.js, lines 9–16

function store(key, value) {
  try {
    if (value === undefined) return window.localStorage.getItem(key) || "";
    window.localStorage.setItem(key, value);
  } catch (e) {
    return "";
  }
}

el builds every element the script adds. Text always goes in through textContent or text nodes, never innerHTML, so nothing a reader types and no setting can become markup.

_extensions/issue-comments/issue-comments.js, lines 18–28

function el(tag, attrs, children) {
  const node = document.createElement(tag);
  Object.entries(attrs || {}).forEach(([k, v]) => {
    if (k === "text") node.textContent = v;
    else if (v !== false && v != null) node.setAttribute(k, v === true ? "" : v);
  });
  (children || []).forEach((c) =>
    node.appendChild(typeof c === "string" ? document.createTextNode(c) : c)
  );
  return node;
}

safeLink accepts the contact setting only as an ordinary web, mail or relative link, so a javascript: address in a site’s settings cannot run when a reader clicks it.

_extensions/issue-comments/issue-comments.js, lines 30–33

// Only allow ordinary links for the contact address.
function safeLink(url) {
  return /^(https?:|mailto:|\/|\.\/|\.\.\/)/i.test(url || "") ? url : null;
}

issueTitle builds the issue title for the preview: the section in brackets, then the first 60 characters of the comment. The relay builds the real title with its own copy of this logic.

_extensions/issue-comments/issue-comments.js, lines 35–40

// Must match buildIssue_ in apps-script/Code.gs.
function issueTitle(section, comment) {
  const flat = comment.replace(/\s+/g, " ");
  const words = flat.slice(0, 60) + (flat.length > 60 ? "…" : "");
  return (section ? "[" + section.slice(0, 60) + "] " : "") + words;
}

The Comment buttons

There is one floating button, moved to each new selection. Pressing a mouse button normally clears the selection, so mousedown is cancelled to keep the highlighted text in place until the click opens the dialog.

  text: "💬 Comment",

Building the dialog

The dialog is a native <dialog> element, which the browser keeps keyboard focus inside and closes with Escape. It is built once and reused. The edit stage has the passage, the name and comment fields, the passcode field when enabled, and the honeypot: a field placed off-screen and hidden from screen readers, so only a bot fills it in.

_extensions/issue-comments/issue-comments.js, lines 76–83

// ---- Dialog: edit stage ----
const quoteBox = el("blockquote", { class: "ic-quote" });
const sectionLine = el("p", { class: "ic-section" });
const nameInput = el("input", { type: "text", name: "name", required: true, maxlength: "80", autocomplete: "name" });
const commentInput = el("textarea", { name: "comment", required: true, maxlength: "5000", rows: "6" });
const passInput = el("input", { type: "password", name: "passcode", autocomplete: "off", required: !!cfg.passcode });
// Honeypot: hidden from people, often filled in by spam bots.
const trap = el("input", { type: "text", name: "website", tabindex: "-1", autocomplete: "off", class: "ic-trap", "aria-hidden": "true" });

Moving between stages

The dialog is always in one of three stages. setStage shows the right parts and relabels the two buttons for each, so the left button is Cancel, then Edit, then Close, and the right is Preview or Post publicly. The left button goes back from the preview, and otherwise closes the dialog.

_extensions/issue-comments/issue-comments.js, lines 128–145

let stage = "edit"; // "edit" | "preview" | "done"

function setStage(s) {
  stage = s;
  editStage.hidden = s !== "edit";
  previewStage.hidden = s !== "preview";
  goBtn.hidden = s === "done";
  goBtn.disabled = false;
  heading.textContent = s === "preview" ? "Check before posting" : "Leave a comment";
  backBtn.textContent = s === "edit" ? "Cancel" : s === "preview" ? "Edit" : "Close";
  goBtn.textContent = s === "edit" && cfg.preview ? "Preview" : "Post publicly";
  if (s !== "done") { status.textContent = ""; status.className = "ic-status"; }
}

backBtn.addEventListener("click", () => {
  if (stage === "preview") setStage("edit");
  else dialog.close();
});

openDialog loads a passage into the dialog and resets it. Focus goes to the comment box when the name is already remembered, and to the name box otherwise.

_extensions/issue-comments/issue-comments.js, lines 147–158

function openDialog(ctx) {
  current = ctx;
  selBtn.hidden = true;
  quoteBox.textContent = ctx.quote;
  quoteBox.hidden = !ctx.quote;
  sectionLine.textContent = ctx.section ? "Section: " + ctx.section : "";
  nameInput.value = store(NAME_KEY);
  commentInput.value = "";
  setStage("edit");
  dialog.showModal();
  (nameInput.value ? commentInput : nameInput).focus();
}

fillPreview draws the issue as it will appear. It mirrors the text buildIssue_ writes on the relay, without the section link.

_extensions/issue-comments/issue-comments.js, lines 160–168

function fillPreview(name, comment) {
  pvTitle.textContent = issueTitle(current.section, comment);
  pvBody.replaceChildren(
    el("p", {}, [el("strong", { text: "Submitted by: " }), name + ", via the site's comment form"]),
    el("p", {}, [el("strong", { text: "Section: " }), current.section || "Page"]),
    ...(current.quote ? [el("blockquote", { class: "ic-quote", text: current.quote })] : []),
    el("p", { class: "ic-pv-comment", text: comment })
  );
}

Posting

post sends the comment and shows the result. The body is JSON, but labelled text/plain: a JSON content type would make the browser send a CORS preflight request first, which Apps Script web apps cannot answer, and the post would be blocked. Apps Script follows a redirect to return its reply, which fetch follows automatically. Any { ok: false } reply is shown to the reader as the relay’s own message.

_extensions/issue-comments/issue-comments.js, lines 170–211

async function post(name, comment) {
  const payload = {
    name,
    comment,
    quote: current.quote,
    section: current.section,
    anchor: current.anchor,
    page: location.origin + location.pathname,
    passcode: passInput.value,
    website: trap.value,
  };
  goBtn.disabled = true;
  backBtn.disabled = true;
  status.textContent = "Posting…";
  status.className = "ic-status";
  try {
    // text/plain avoids a CORS preflight, which Apps Script cannot answer.
    const res = await fetch(cfg.endpoint, {
      method: "POST",
      headers: { "Content-Type": "text/plain;charset=utf-8" },
      body: JSON.stringify(payload),
    });
    const data = await res.json();
    if (!data.ok) throw new Error(data.error || "The comment was not posted.");
    setStage("done");
    status.className = "ic-status ic-ok";
    status.replaceChildren(
      "Thanks! Posted as ",
      data.url
        ? el("a", { href: data.url, target: "_blank", rel: "noopener", text: "issue #" + data.number })
        : "an issue",
      "."
    );
    window.getSelection().removeAllRanges();
  } catch (err) {
    status.textContent = err.message || "Something went wrong. Please try again.";
    status.className = "ic-status ic-error";
    goBtn.disabled = false;
  } finally {
    backBtn.disabled = false;
  }
}

The form’s submit handler runs the browser’s own validation of the required fields, tidies the name, remembers it, and then either shows the preview or posts.

_extensions/issue-comments/issue-comments.js, lines 213–227

form.addEventListener("submit", (e) => {
  e.preventDefault();
  if (stage === "edit" && !form.reportValidity()) return;
  const name = nameInput.value.trim().replace(/\s+/g, " ");
  const comment = commentInput.value.trim();
  if (!name || !comment) return;
  store(NAME_KEY, name);
  if (stage === "edit" && cfg.preview) {
    fillPreview(name, comment);
    setStage("preview");
    goBtn.focus();
  } else {
    post(name, comment);
  }
});

Tracking the selection

sectionFor finds the section a passage belongs to. Quarto wraps each heading and its content in a <section> with the heading’s id, so the nearest enclosing section[id] gives both the heading text and the anchor for the issue’s link.

_extensions/issue-comments/issue-comments.js, lines 230–238

function sectionFor(node) {
  const start = node.nodeType === 1 ? node : node.parentElement;
  const section = start && start.closest("section[id]");
  if (section) {
    const h = section.querySelector(":scope > h1, :scope > h2, :scope > h3, :scope > h4, :scope > h5, :scope > h6");
    return { section: h ? h.textContent.trim() : section.id, anchor: section.id };
  }
  return { section: document.title, anchor: "" };
}

update decides whether to show the button and where. It ignores selections while the dialog is open, ones shorter than two characters, and ones outside the main content. The button goes just under the last line of the selection, kept inside the window.

_extensions/issue-comments/issue-comments.js, lines 240–265

function update() {
  if (dialog.open) return;
  const sel = window.getSelection();
  if (!sel || sel.isCollapsed || sel.rangeCount === 0) {
    selBtn.hidden = true;
    return;
  }
  const range = sel.getRangeAt(0);
  const text = sel.toString().replace(/\s+\n/g, "\n").trim();
  if (text.length < 2 || !root.contains(range.commonAncestorContainer)) {
    selBtn.hidden = true;
    return;
  }
  const info = sectionFor(range.startContainer);
  current = {
    quote: text.length > MAX_QUOTE ? text.slice(0, MAX_QUOTE) + "…" : text,
    section: info.section,
    anchor: info.anchor,
  };
  const rects = range.getClientRects();
  const rect = rects.length ? rects[rects.length - 1] : range.getBoundingClientRect();
  const maxLeft = document.documentElement.clientWidth - 140;
  selBtn.style.top = rect.bottom + window.scrollY + 6 + "px";
  selBtn.style.left = Math.max(8, Math.min(rect.right + window.scrollX - 60, maxLeft)) + "px";
  selBtn.hidden = false;
}

selectionchange fires many times while a reader drags, so update runs only once the selection has been still for 120 milliseconds. Resizing the window reflows the text, so it repositions the button too.

_extensions/issue-comments/issue-comments.js, lines 267–273

let timer = null;
const schedule = () => {
  clearTimeout(timer);
  timer = setTimeout(update, 120);
};
document.addEventListener("selectionchange", schedule);
window.addEventListener("resize", schedule);
